                id             INTEGER PRIMARY KEY AUTOINCREMENT,

                leg_id         INTEGER,

                event_status   TEXT    NOT NULL DEFAULT 'registered',
                event_date     TEXT    NOT NULL,
                quantity       INTEGER,
