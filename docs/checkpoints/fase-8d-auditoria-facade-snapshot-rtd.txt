    44	def _get_structure_info(structure_id: int, db_path: Path) -> tuple[str, str]:
    45	    """
    46	    Retorna (alias_legacy_aba, underlying_asset) para a estrutura.
    47	
    48	    Raises ValueError se:
    49	      - estrutura não existir
    50	      - alias_legacy_aba for nulo (sem aba legada mapeada)
    51	    """
    52	    with sqlite3.connect(str(db_path)) as conn:
    53	        conn.row_factory = sqlite3.Row
    54	        row = conn.execute(
    55	            "SELECT alias_legacy_aba, underlying_asset FROM structures WHERE id = ?",
    56	            (structure_id,),

    59	    if row is None:
    60	        raise ValueError(f"structure not found: {structure_id}")
    61	
    62	    aba = row["alias_legacy_aba"]
    63	    if not aba:
    64	        raise ValueError(f"alias_legacy_aba is null for structure_id={structure_id}")
    65	
    66	    underlying_asset = row["underlying_asset"]  # NOT NULL -- sempre presente
    67

    75	def _to_float(value: Any, default: float = 0.0) -> float:
    76	    try:
    77	        if value is None or value == "":
    78	            return default
    79	
    80	        if isinstance(value, str):
    81	            text = value.strip()
    82	            text = text.replace("R$", "").replace("$", "").strip()
    83	
    84	            # Remove espaços internos comuns em valores monetários.
    85	            text = text.replace(" ", "")
    86	
    87	            # Caso BR simples: "124,66"
    88	            if "," in text and "." not in text:
    89	                text = text.replace(",", ".")
    90	
    91	            return float(text)
    92	
    93	        return float(value)
    94	    except Exception:
    95	        return default

    98	def _normalize_expiration_date(value: Any) -> str | None:
    99	    if not value:
   100	        return None
   101	
   102	    text = str(value).strip()
   103	
   104	    formats = [
   105	        "%m/%d/%Y %H:%M",
   106	        "%m/%d/%Y %H:%M:%S",
   107	        "%Y-%m-%d",
   108	        "%Y-%m-%dT%H:%M:%S",
   109	    ]
   110	
   111	    for fmt in formats:
   112	        try:
   113	            return datetime.strptime(text, fmt).date().isoformat()
   114	        except ValueError:
   115	            pass
   116	
   117	    return text

   132	def _lookup_spot_price(db_path: Path, underlying_asset: str) -> float:
   133	    """
   134	    Procura spot positivo no app.db.
   135	
   136	    Caso confirmado:
   137	      estrutura SMAL11 possui spot positivo disponível na base canônica/staging.
   138	      spot observado = 124.66
   139	    """
   140	    if not underlying_asset:
   141	        return 0.0
   142	
   143	    symbol_candidates = {
   144	        "aba",
   145	        "ativo",
   146	        "asset",
   147	        "symbol",
   148	        "ticker",
   149	        "underlying_asset",
   150	        "codigo",
   151	        "papel",
   152	    }
   153	
   154	    price_candidates = {
   155	        "spot",
   156	        "spot_price",
   157	        "underlying_price",
   158	        "last_price",
   159	        "price",
   160	        "preco",
   161	        "preco_atual",
   162	        "valor",
   163	        "cotacao",
   164	        "ultimo",
   165	        "fechamento",
   166	        "close",
   167	    }
   168	
   169	    try:
   170	        with sqlite3.connect(str(db_path)) as conn:
   171	            tables = conn.execute(
   172	                "SELECT name FROM sqlite_master WHERE type = 'table'"
   173	            ).fetchall()
   174	
   175	            for (table_name,) in tables:
   176	                columns_info = conn.execute(
   177	                    f"PRAGMA table_info({_quote_ident(table_name)})"
   178	                ).fetchall()
   179	
   180	                columns = [row[1] for row in columns_info]
   181	                lower_to_real = {col.lower(): col for col in columns}
   182	
   183	                symbol_cols = [
   184	                    lower_to_real[name]
   185	                    for name in symbol_candidates
   186	                    if name in lower_to_real
   187	                ]
   188	
   189	                price_cols = [
   190	                    lower_to_real[name]
   191	                    for name in price_candidates
   192	                    if name in lower_to_real
   193	                ]
   194	
   195	                if not symbol_cols or not price_cols:
   196	                    continue
   197	
   198	                for symbol_col in symbol_cols:
   199	                    for price_col in price_cols:
   200	                        query = (
   201	                            f"SELECT {_quote_ident(price_col)} "
   202	                            f"FROM {_quote_ident(table_name)} "
   203	                            f"WHERE UPPER(CAST({_quote_ident(symbol_col)} AS TEXT)) = UPPER(?) "
   204	                            f"AND {_quote_ident(price_col)} IS NOT NULL "
   205	                            f"LIMIT 20"
   206	                        )
   207	
   208	                        try:
   209	                            rows = conn.execute(query, (underlying_asset,)).fetchall()
   210	                        except Exception:
   211	                            continue
   212	
   213	                        for row in rows:
   214	                            price = _to_float(row[0], 0.0)
   215	                            if price > 0:
   216	                                return price
   217	    except Exception:
   218	        return 0.0
   219	
   220	    return 0.0

   223	def _snapshot_result_to_payload(
   224	    selection_result: Any,
   225	    structure_id: int,
   226	    underlying_asset: str,
   227	    reference_date: str | None,
   228	    db_path: Path,
   229	) -> dict[str, Any]:
   230	    legs_data = []
   231	
   232	    for leg in selection_result.legs:
   233	        d = leg if isinstance(leg, dict) else vars(leg)
   234	
   235	        quantity = _to_float(_pick(d, "quantity", "quant"), 0.0)
   236	
   237	        raw_price = _pick(d, "premium", "price", "valor_executado")
   238	        raw_asset = _pick(d, "symbol", "asset", "ativo")
   239	        raw_expiry = _pick(d, "expiration_date", "expiry", "vencimento")
   240	
   241	        side = _pick(d, "side", "position_side")
   242	        if not side:
   243	            side = "SHORT" if quantity < 0 else "LONG"
   244	
   245	        canonical_leg = {
   246	            # campos originais/compatíveis
   247	            "quantity":    quantity,
   248	            "price":       _to_float(raw_price, 0.0),
   249	            "asset":       raw_asset,
   250	            "option_type": _pick(d, "option_type", "call_put"),
   251	            "strike":      _to_float(_pick(d, "strike"), 0.0),
   252	            "expiry":      raw_expiry,
   253	            "iv":          _pick(d, "iv"),
   254	            "delta":       _pick(d, "delta"),
   255	            "gamma":       _pick(d, "gamma"),
   256	            "theta":       _pick(d, "theta"),
   257	            "vega":        _pick(d, "vega"),
   258	            "source":      str(_pick(d, "source")),
   259	
   260	            # campos canônicos esperados pelo fluxo pricing/payoff
   261	            "symbol":          raw_asset,
   262	            "premium":         _to_float(raw_price, 0.0),
   263	            "expiration_date": _normalize_expiration_date(raw_expiry),
   264	            "multiplier":      1.0,
   265	            "side":            str(side).upper(),
   266	            "position_side":   str(side).upper(),
   267	        }
   268	
   269	        legs_data.append(canonical_leg)
   270	
   271	    spot = (
   272	        getattr(selection_result, "spot_price", None)
   273	        or getattr(selection_result, "spot", None)
   274	        or getattr(selection_result, "underlying_price", None)
   275	        or getattr(selection_result, "last_price", None)
   276	    )
   277	
   278	    spot_price = _to_float(spot, 0.0)
   279	
   280	    if spot_price <= 0:
   281	        spot_price = _lookup_spot_price(
   282	            db_path=db_path,
   283	            underlying_asset=underlying_asset,
   284	        )
   285	
   286	    if spot_price <= 0:
   287	        raise ValueError(
   288	            f"spot_price inválido ou ausente para underlying_asset={underlying_asset}. "
   289	            "Não persistir execução OK com spot_price <= 0."
   290	        )
   291	
   292	    return {
   293	        "structure_id":     structure_id,
   294	        "underlying_asset": underlying_asset,
   295	        "reference_date":   reference_date,
   296	        "spot_price":       spot_price,
   297	        "interest_rate":    0.0,
   298	        "volatility":       0.0,
   299	        "legs":             legs_data,
   300	        "meta": {
   301	            "snapshot_source":  str(selection_result.source),
   302	            "snapshot_aba":     selection_result.aba,
   303	            "manual_overrides": getattr(selection_result, "manual_overrides", None) or [],
   304	            "legs_count":       len(legs_data),
   305	        },
   306	    }

   309	class CanonicalPricingFacade:
   310	    """
   311	    Orquestra o pipeline canônico ponta a ponta:
   312	
   313	        structure_id
   314	             alias_legacy_aba + underlying_asset  (query em structures)
   315	                     MarketSnapshotSelector.select(aba=...)
   316	                             pricing_payload  (underlying_asset = ativo real)
   317	                                     PricingExecutionService.execute_payload()
   318	                                             PricingExecutionPersistenceService.persist()
   319	                                                     DerivedPayoffPersistence.persist()
   320	                                                             derived.db
   321	    """
   322	
   323	    def __init__(
   324	        self,
   325	        db_path: Path | str = _DEFAULT_DB,
   326	        pricing_execution_service: PricingExecutionService | None = None,
   327	        persistence_service: PricingExecutionPersistenceService | None = None,
   328	    ) -> None:
   329	        self._db_path  = Path(db_path)
   330	        self._repo     = MarketSnapshotRepository(db_path=self._db_path)
   331	        self._selector = MarketSnapshotSelector(repository=self._repo)
   332	        self._engine   = pricing_execution_service or PricingExecutionService()
   333	
   334	        self._persister = persistence_service or PricingExecutionPersistenceService(
   335	            payoff_persistence_port=DerivedPayoffPersistence(),
   336	            system_snapshots_repository=SystemSnapshotsRepository(db_path=self._db_path),
   337	        )
   338	
   339	    def execute_pricing(
   340	        self,
   341	        structure_id: int,
   342	        reference_date: str | None = None,
   343	    ) -> dict[str, Any]:
   344	        started_at = time.perf_counter()
   345	
   346	        try:
   347	            #  1. Resolve aba + underlying_asset 
   348	            aba, underlying_asset = _get_structure_info(   #  C6/C8
   349	                structure_id, self._db_path
   350	            )
   351	
   352	            #  2. Seleciona snapshot (manual > rtd) 
   353	            selection = self._selector.select(aba=aba)
   354	
   355	            #  3. Monta pricing_payload 
   356	            pricing_payload = _snapshot_result_to_payload(
   357	                selection_result=selection,
   358	                structure_id=structure_id,
   359	                underlying_asset=underlying_asset,          #  C7/C8
   360	                reference_date=reference_date,
   361	                db_path=self._db_path,
   362	            )
   363	
   364	            #  4. Executa engine 
   365	            execution_result = self._engine.execute_payload(
   366	                pricing_payload=pricing_payload,
   367	            )
   368	
   369	            #  C4: extrai dict interno do wrapper 
   370	            engine_result = execution_result.get("result", execution_result)
   371	
   372	            duration_ms = int((time.perf_counter() - started_at) * 1000)
   373	
   374	            #  5. Persiste (app.db + derived.db via port) 
   375	            persisted = self._persister.persist_execution(
   376	                pricing_payload=pricing_payload,
   377	                result=engine_result,
   378	                duration_ms=duration_ms,
   379	                error_message=None,
   380	            )
   381	
   382	            return {
   383	                "status":          "ok",
   384	                "canonical_input": pricing_payload,
   385	                "pricing_payload": pricing_payload,
   386	                "result":          execution_result,
   387	                "persisted":       persisted,
   388	                "meta":            pricing_payload["meta"],
   389	                "duration_ms":     duration_ms,
   390	            }
   391	
   392	        except Exception as exc:
   393	            duration_ms   = int((time.perf_counter() - started_at) * 1000)
   394	            error_message = str(exc)
   395	
   396	            try:
   397	                self._persister.persist_execution(
   398	                    pricing_payload=None,
   399	                    result={"engine": "stub", "status": "error", "error_message": error_message},
   400	                    duration_ms=duration_ms,
   401	                    error_message=error_message,
   402	                )
   403	            except Exception:
   404	                pass
   405	
   406	            return {
   407	                "status":          "error",
   408	                "canonical_input": None,
   409	                "pricing_payload": None,
   410	                "result":          None,
   411	                "persisted":       None,
   412	                "meta":            {},
   413	                "duration_ms":     duration_ms,
   414	                "error_message":   error_message,
   415	            }

   138	    # Resolução de snapshot — alteracao_25: duas responsabilidades separadas
   139	    # ──────────────────────────────────────────────────────────────────────────
   140	
   141	    def _resolve_snapshot(
   142	        self,
   143	        structure: dict[str, Any],
   144	        reference_date: str | None,
   145	    ) -> tuple[dict[str, Any], dict[str, Any]]:
   146	        """
   147	        Monta o snapshot dict completo exigido pelo assembler.
   148	
   149	        Sempre busca spot/taxa/vol no MarketSnapshotProvider (fonte autoritativa).
   150	        Se o selector estiver injetado E alias_legacy_aba existir, substitui as
   151	        legs pelo resultado do selector (manual > rtd).
   152	        """
   153	        underlying_asset = structure["underlying_asset"]
   154	        aba              = structure.get("alias_legacy_aba")
   155	
   156	        # 1. Spot, taxa, vol — sempre via provider
   157	        base_snapshot = self.market_snapshot_provider.get_snapshot(
   158	            underlying_asset,
   159	            reference_date=reference_date,
   160	        )
   161	
   162	        # 2. Legs — via selector se disponível, senão mantém o que o provider trouxe
   163	        if self.market_snapshot_selector is not None and aba:
   164	            ref = StructureRef.from_aba(aba)
   165	            legs_list, legs_meta = self._resolve_legs_via_selector(ref)
   166	            snapshot_source = legs_meta["snapshot_source"]
   167	        else:
   168	            legs_list  = base_snapshot.get("legs", [])
   169	            legs_meta  = {}
   170	            snapshot_source = "provider_legacy"
   171	
   172	        # 3. Monta contrato completo para o assembler
   173	        snapshot = {
   174	            **base_snapshot,            # reference_date, underlying_asset, spot_price,
   175	                                        # interest_rate, volatility (e qualquer extra)
   176	            "aba":  aba,
   177	            "legs": legs_list,
   178	        }
   179	
   180	        meta = {
   181	            "snapshot_source":  snapshot_source,
   182	            **legs_meta,
   183	        }
   184	
   185	        return snapshot, meta
   186	
   187	    # ──────────────────────────────────────────────────────────────────────────
   188	    # Legs via selector (manual > rtd)
   189	    # ──────────────────────────────────────────────────────────────────────────
   190	
   191	    def _resolve_legs_via_selector(
   192	        self,
   193	        ref: StructureRef,
   194	    ) -> tuple[list[dict[str, Any]], dict[str, Any]]:
   195	        """
   196	        Delega ao MarketSnapshotSelector e serializa legs completas.
   197	
   198	        Serialização cobre todos os campos de LegMarketSnapshot para que
   199	        consumidores downstream (pricing, greeks, payoff) tenham os dados.
   200	        """
   201	        aba_str = ref.aba
   202	        result = self.market_snapshot_selector.select(aba_str)
   203	
   204	        legs_as_dict = [
   205	            {
   206	                # ── identificação ──────────────────────────────────────────
   207	                "aba":              leg.aba,
   208	                "ativo":            leg.ativo,
   209	                "source":           leg.source.value if hasattr(leg.source, "value") else str(leg.source),
   210	                # ── posição ────────────────────────────────────────────────
   211	                "cv":               leg.cv,
   212	                "call_put":         leg.call_put,
   213	                "quant":            leg.quant,
   214	                "valor_executado":  leg.valor_executado,
   215	                # ── preços ─────────────────────────────────────────────────
   216	                "bid":              leg.bid,
   217	                "ask":              leg.ask,
   218	                "mid":              leg.mid,
   219	                "spread":           leg.spread,
   220	                "spread_pct":       leg.spread_pct,
   221	                # ── greeks ─────────────────────────────────────────────────
   222	                "iv":               leg.iv,
   223	                "delta":            leg.delta,
   224	                "gamma":            leg.gamma,
   225	                "theta":            leg.theta,
   226	                "vega":             leg.vega,
   227	                # ── contrato ───────────────────────────────────────────────
   228	                "strike":           leg.strike,
   229	                "vencimento":       leg.vencimento,
   230	                "dte":              leg.dte,
   231	                "pl_realista":      leg.pl_realista,
   232	                # ── auditoria ──────────────────────────────────────────────
   233	                "timestamp":        leg.timestamp,
   234	            }
   235	            for leg in result.legs
   236	        ]
   237	
   238	        # reference_date derivada do timestamp da leg mais recente
   239	        reference_date = self._reference_date_from_legs(result.legs)
   240	
   241	        meta = {
   242	            "snapshot_source":  result.source.value if hasattr(result.source, "value") else str(result.source),
   243	            "snapshot_aba":     aba_str,
   244	            "manual_overrides": result.manual_overrides,
   245	            "is_manual_first":  result.is_manual_first,
   246	            "legs_reference_date": reference_date,
   247	        }
   248	
   249	        return legs_as_dict, meta

   251	    @staticmethod
   252	    def _reference_date_from_legs(legs) -> str | None:
   253	        """Extrai a data (YYYY-MM-DD) do timestamp mais recente entre as legs."""
   254	        timestamps = [leg.timestamp for leg in legs if leg.timestamp]
   255	        if not timestamps:
   256	            return None
   257	        latest = max(timestamps)          # ISO string → max() funciona diretamente
   258	        return latest[:10]               # "YYYY-MM-DD HH:MM:SS" → "YYYY-MM-DD"

   116	def _parse_br_float(value) -> Optional[float]:
   117	    # Converte string pt-BR ('1,38' ou '1,38E-02') para float.
   118	    if value is None:
   119	        return None
   120	    try:
   121	        normalized = str(value).strip().replace(",", ".")
   122	        return float(normalized)
   123	    except (ValueError, TypeError):
   124	        return None

   127	def _mid_price(bid: Optional[float], ask: Optional[float]) -> Optional[float]:
   128	    # Calcula mid price. Nao usa coluna 'last' - nao existe no schema.
   129	    if bid is not None and ask is not None:
   130	        return round((bid + ask) / 2.0, 6)
   131	    if bid is not None:
   132	        return bid
   133	    if ask is not None:
   134	        return ask
   135	    return None

   139	def _sort_rows_by_timestamp_desc(rows: list[sqlite3.Row]) -> list[sqlite3.Row]:
   140	    """
   141	    Ordena rows por timestamp cronológico decrescente.
   142	
   143	    Evita ordenar pelo campo textual de data no SQLite, pois o banco legado pode misturar
   144	    formatos ISO e BR, fazendo a ordenação textual ficar incorreta.

   147	        raw = "" if row["timestamp"] is None else str(row["timestamp"]).strip()
   148	
   149	        try:
   150	            return (1, parse_timestamp(raw), raw)
   151	        except Exception:
   152	            return (0, datetime.min, raw)
   153	
   154	    return sorted(rows, key=_key, reverse=True)

   192	    """
   193	    Acesso de leitura aos snapshots de mercado.
   194	
   195	    Metodos:
   196	      get_rtd_legs(aba)     -> lista de LegMarketSnapshot source=RTD
   197	      get_manual_legs(aba)  -> lista de LegMarketSnapshot source=MANUAL

   202	    def __init__(self, db_path: Path | str = _DEFAULT_DB) -> None:
   203	        self._db_path = Path(db_path)

   205	            raise FileNotFoundError(f"Banco nao encontrado: {self._db_path}")

   208	        conn = sqlite3.connect(str(self._db_path))

   218	        rows = _sort_rows_by_timestamp_desc(rows)
   219	        return [_row_to_leg(r, SnapshotSource.RTD) for r in rows]

   222	        aba = _ref_to_aba(ref)
   223	        with self._connect() as conn:
   224	            row = conn.execute(_SQL_RTD_SUMMARY, (aba,)).fetchone()
   225	        if row is None:
   226	            return None
   227	        return dict(row)
