     4	def _clean_text(value: Any) -> str | None:
     5	    if value is None:
     6	        return None
     7	    text = str(value).strip()
     8	    return text or None
     9	
    10	
    11	def _clean_upper_text(value: Any) -> str | None:
    12	    text = _clean_text(value)
    13	    return text.upper() if text is not None else None
    14	
    15	
    16	def to_pricing_payload(canonical_input: dict[str, Any]) -> dict[str, Any]:
    17	    if not canonical_input:
    18	        raise ValueError("canonical_input is required")
    19	
    20	    structure = canonical_input.get("structure")
    21	    market = canonical_input.get("market")
    22	
    23	    if not structure:
    24	        raise ValueError("canonical_input.structure is required")
    25	
    26	    if not market:
    27	        raise ValueError("canonical_input.market is required")
    28	
    29	    legs = structure.get("legs", [])
    30	    pricing_legs = []
    31	
    32	    for index, leg in enumerate(legs):
    33	        if not leg:
    34	            raise ValueError(f"canonical_input.structure.legs[{index}] is required")
    35	
    36	        pricing_legs.append(
    37	            {
    38	                "side": _clean_upper_text(leg["position_side"]),
    39	                "instrument_type": "OPTION",
    40	                "option_type": _clean_upper_text(leg["option_type"]),
    41	                "symbol": _clean_upper_text(leg.get("symbol")),
    42	                "strike": float(leg["strike"]),
    43	                "expiration_date": _clean_text(leg["expiration_date"]),
    44	                "quantity": int(leg["quantity"]),
    45	                "premium": float(leg["premium"]) if leg.get("premium") is not None else None,
    46	                "multiplier": float(leg["multiplier"]),
    47	            }
    48	        )
    49	
    50	    return {
    51	        "structure_id": structure["structure_id"],
    52	        "structure_name": _clean_text(structure["name"]),
    53	        "underlying_asset": _clean_upper_text(structure["underlying_asset"]),
    54	        "reference_date": _clean_text(market["reference_date"]),
    55	        "spot_price": float(market["spot_price"]),
    56	        "interest_rate": float(market["interest_rate"]),
    57	        "volatility": float(market["volatility"]),
    58	        "legs": pricing_legs,
    59	    }
